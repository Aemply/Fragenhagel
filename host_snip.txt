function renderHostBuzz(){

  const metaEl=document.getElementById('hostBuzzMeta');
  if(!nameEl || !metaEl) return;
  const current=state?.buzzedBy || null;

  // The visible indication is only for the currently active buzzer.
  // After Richtig/Falsch, buzzedBy is cleared, so the indication disappears.
  if(!current){
    nameEl.textContent='Noch niemand gebuzzert';
    nameEl.className='host-buzz-name waiting';
    metaEl.textContent=state?.buzzerOpen ? 'Buzzer ist geöffnet' : 'Buzzer ist geschlossen';
    return;
  }

  nameEl.textContent='🔔 '+current;
  nameEl.className='host-buzz-name active';
  metaEl.textContent = 'Hat gerade gebuzzert';
}

function renderJudge(points){
 const j=document.getElementById('judge');
 if(!j)return;
 const currentBuzz=state?.buzzedBy || null;
 j.innerHTML='';

 // Absolute rule: before somebody buzzes, there are NO judgement buttons.
 if(!currentBuzz)return;

 const ps=state.players||[];
 const p=ps.find(x=>String(x.name)===String(currentBuzz));
 if(!p)return;

 const val=Number(points)||100;

 const addBtn=(label, cls, fn)=>{
   const b=document.createElement('button');
   b.className='btn '+cls;
   b.type='button';
   b.textContent=label;
   b.dataset.player=String(p.name);
   b.disabled=!!judgedPlayers[String(p.name)];
   b.addEventListener('click',()=>{ if(!b.disabled) fn(); });
   j.appendChild(b);
 };

 if(active?.type==='Face Morph'){
   const hasPreviousWrong=Array.isArray(state?.lockedPlayers) && state.lockedPlayers.length>0;
   if(hasPreviousWrong){
     addBtn('👤 '+String(p.name)+' +100','good',()=>faceMorphSecondCorrect(p.name));
     addBtn('❌ Falsch – '+String(p.name),'bad',()=>faceMorphSecondWrong(p.name));
   }else{
     addBtn('👤 '+String(p.name)+' +100','good',()=>award(p.name,100));
     addBtn('👥 '+String(p.name)+' +300','good',()=>award(p.name,300));
   }
 }else{
   addBtn('✅ Richtig – '+String(p.name),'good',()=>correct(p.name,val));
   addBtn('❌ Falsch – '+String(p.name),'bad',()=>wrong(p.name,val));
 }
}
function lockPlayerButtons(p){
 const key=String(p);
 judgedPlayers[key]=true;
 const j=document.getElementById('judge');
 if(!j)return;
 // Disable only the two judgement buttons belonging to this player.
 // Other players remain fully clickable after a wrong answer.
 const buttons=[...j.querySelectorAll('button')];
 buttons.filter(b=>b.dataset.player===key).forEach(b=>b.disabled=true);
}

function award(p,v){
 if(judgedPlayers[String(p)])return;
 lockPlayerButtons(p);
 const pts=Number(v)||0;
 if(pts===100){
   // Face Morph: +100 is still a wrong answer.
   // The player keeps the 100 points, but is locked out and the buzzer reopens for others.
   try{sWrong()}catch(e){}
   socket.emit('faceMorphPartial',{player:String(p),points:100});
 }else if(pts===300){
   // Face Morph: +300 ends the question and automatically reveals the solution.
   try{sCorrect()}catch(e){}
   socket.emit('correct',{player:String(p),points:300});
   setTimeout(()=>{
     try{
       if(typeof reveal === 'function') reveal();
     }catch(e){
       console.error('Automatisches Anzeigen der Lösung fehlgeschlagen:',e);
     }
   },250);
 }else{
   try{sCorrect()}catch(e){}
   socket.emit('correct',{player:String(p),points:pts});
 }
}
